public class DoubleTimedValue(double value, double time) {
    public DoubleTimedValue(double value, double time){
        this.value = value;
        this.time = time;
    }
    public void setValue(double value){
        this.value = value;
    }
    public void setTime(double time){
            this.time = time;
    }
}

import java.time.LocalTime;


public class DoubleDataSender {


   private final String name;
   private DoubleTimedValue doubleTimedValue;


   public DoubleDataSender(String name) {
       this.name = name;
       this.doubleTimedValue = new DoubleTimedValue(0,0);
   }


   public void update() {
       this.doubleTimedValue.setValue(Math.random());
       this.doubleTimedValue.setTime(LocalTime.now().getSecond());
   }


   public DoubleTimedValue getDoubleTimedValue() {
       return doubleTimedValue;
   }


}
